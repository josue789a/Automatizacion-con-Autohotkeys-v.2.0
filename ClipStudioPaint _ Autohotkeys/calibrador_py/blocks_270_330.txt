    def _reset(self, name, ent):
        ev = self.ctx.catalogo.por_id(self.ctx.event_id)
        def_val = (ev.atajo_def if name == "atajo" else ev.remap_def) if ev else ""
        def_val = "" if def_val == "—" else def_val
        ent.delete(0, "end")
        ent.insert(0, def_val)
        self._commit(name, "")
        self._check_dup()
