1. Saturación de la cola de mensajes

PlaySoundW con SND_MEMORY|SND_ASYNC no encola nada. Cada llamada reemplaza el sonido anterior, sin hilos, sin SetTimer y sin I/O.
No uses SetTimer(...,-1) en la ruta granular: eso sí llenaría la cola del hilo AHK.
Hay tres frenos contra la saturación:
Cuantización: solo suena cada 3 unidades, así que un barrido completo da ≤ 34 clics.
Mínimo de 12 ms entre sonidos.
El bucle de arrastre ya hace Sleep(8).
Los 11 buffers (uno por banda de 10 %, 650–850 Hz, 7 ms, unos 350 bytes cada uno) se precargan al arrancar. La latencia restante es la del buffer de WinMM, no la de AHK.

2. Racha en Ctrl+Z / Ctrl+Y

Hay 5 niveles pentatónicos mayores (razones 1, 9/8, 5/4, 3/2, 5/3). Undo parte de 520 Hz y Redo de 760 Hz.
Si pulsas en menos de 400 ms, el nivel sube hasta el tope (sin volver a empezar).
Si te detienes, el nivel baja uno por cada 500 ms de pausa pasados los 400 ms. Con unos 1,9 s de inactividad vuelve a cero.

3. Especificaciones de audio

Tonos RAM: 22.050 Hz, mono, PCM 16-bit, 6–24 ms.
WAV: 44.100 Hz, mono, PCM 16-bit, ≤ 150 ms (unos 13 KB). Así el escalado de volumen funciona; PCM float, ADPCM o 24-bit se reproducen pero sin escalar.
fill_loud.wav (354 KB) y recycle_loud.wav (112 KB): probablemente duran más de 150 ms. PlaySound es monofónico y su cola se escucha aunque llegue otro sonido, así que recórtalos.
Texturas nuevas: si añades Wood Block, Mechanical Switch o UI Pop, que sean WAV con esas especificaciones y nombres libres. _SonarWav("nombre", [f0,f1,ms]) cae solo a un tono RAM si el archivo falta.

_AudCache() {
    static c := Map()
    return c
}
_AudActivo() => (_Tema("audio","enabled","1") != "0")
_AudVol() {
    v := _Tema("audio","volume","100")
    return IsInteger(v) ? Max(0,Min(100,Integer(v))) / 100 : 1.0
}
; SND_ASYNC(1)|SND_NODEFAULT(2)|SND_MEMORY(4): sin hilos, sin cola, un sonido reemplaza al anterior
_AudPlay(buf) => DllCall("winmm\PlaySoundW","Ptr",buf,"Ptr",0,"UInt",0x0007)

_AudTonoBuf(id, f0, f1, ms, vol := 0.20) {
    c := _AudCache()
    if !c.Has(id)
        c[id] := _CrearTonoRAM(f0, f1, ms, vol * _AudVol())
    return c[id]
}
_AudTonoRAM(id, f0, f1, ms, vol := 0.20) => _AudPlay(_AudTonoBuf(id, f0, f1, ms, vol))

_AudWavBuf(nombre) {
    c := _AudCache()
    k := "wav:" nombre
    if c.Has(k)
        return c[k]
    ruta := A_ScriptDir "\sonidos\" nombre ".wav"
    buf := 0
    if FileExist(ruta) {
        try {
            buf := FileRead(ruta, "RAW")
            _AudEscalarWav(buf, _AudVol())
        } catch
            buf := 0
    }
    c[k] := buf          ; caché negativa: si falta, no se reintenta el disco
    return buf
}
_AudEscalarWav(buf, vol) {
    if (vol >= 0.995 || buf.Size < 44)
        return
    if (NumGet(buf,0,"UInt") != 0x46464952 || NumGet(buf,8,"UInt") != 0x45564157)
        return
    pos := 12, bits := 0, tag := 0
    while (pos + 8 <= buf.Size) {
        id := NumGet(buf,pos,"UInt")
        sz := NumGet(buf,pos + 4,"UInt")
        if (id = 0x20746D66) {
            tag := NumGet(buf,pos + 8,"UShort")
            bits := NumGet(buf,pos + 22,"UShort")
        } else if (id = 0x61746164) {
            if (tag != 1 || bits != 16)
                return
            n := Min(sz, buf.Size - pos - 8) // 2
            p := pos + 8
            loop n {
                NumPut("Short", Round(NumGet(buf,p,"Short") * vol), buf, p)
                p += 2
            }
            return
        }
        pos += 8 + sz + (sz & 1)
    }
}

_SliderBuf(banda) {
    f := 650 + banda * 20                          ; ~750 Hz de centro, seco y ultracorto
    return _AudTonoBuf("hap_sl_" banda, f, f, 7, 0.24)
}
_HapticoSlider(v) {
    static ultV := -999, ultSnd := 0, ultCall := 0
    if !_AudActivo()
        return
    ahora := A_TickCount
    if (ahora - ultCall > 300)                     ; arrastre nuevo: el primer clic siempre suena
        ultV := -999
    ultCall := ahora
    if (Abs(v - ultV) < 3 || ahora - ultSnd < 12)  ; cada 3 unidades y mínimo 12 ms
        return
    ultV := v
    ultSnd := ahora
    _AudPlay(_SliderBuf(Min(10, Integer(v) // 10)))
}

_RachaBuf(clave, nivel) {
    static r := [1.0, 1.125, 1.25, 1.5, 1.667]     ; pentatónica mayor
    f := Round((clave = "redo" ? 760 : 520) * r[nivel + 1])
    return _AudTonoBuf("rch_" clave "_" nivel, f, Round(f * 1.05), 16, 0.22)
}
_HapticoRacha(clave) {
    static nivel := Map("undo",0,"redo",0), ult := Map("undo",0,"redo",0)
    if !_AudActivo()
        return
    ahora := A_TickCount
    gap := ahora - ult[clave]
    if (gap < 400)
        nivel[clave] := Min(4, nivel[clave] + 1)
    else                                           ; reseteo suave: un nivel por cada 500 ms de pausa
        nivel[clave] := Max(0, nivel[clave] - ((gap - 400) // 500 + 1))
    ult[clave] := ahora
    _AudPlay(_RachaBuf(clave, nivel[clave]))
}

_AudPrecargar() {
    Loop 11
        _SliderBuf(A_Index - 1)
    for clave in ["undo","redo"] {
        Loop 5
            _RachaBuf(clave, A_Index - 1)
    }
    for nombre in ["alpha_loud","alpha_short","fill_loud","recycle_loud"]
        _AudWavBuf(nombre)
}

_GdipRes() {
    static r := {fam:0, fnt:0, fmt:0, tam:0, tok:0}
    return r
}
_GdipLiberarFuentes() {
    gr := _GdipRes()
    if gr.fnt
        DllCall("gdiplus\GdipDeleteFont","ptr",gr.fnt,"int")
    if gr.fmt
        DllCall("gdiplus\GdipDeleteStringFormat","ptr",gr.fmt,"int")
    if gr.fam
        DllCall("gdiplus\GdipDeleteFontFamily","ptr",gr.fam,"int")
    gr.fnt := 0, gr.fmt := 0, gr.fam := 0, gr.tam := 0
}
_AudAlSalir(*) {
    DllCall("winmm\PlaySoundW","Ptr",0,"Ptr",0,"UInt",0)   ; corta el sonido antes de liberar buffers
    _GdipLiberarFuentes()
    gr := _GdipRes()
    if gr.tok {
        DllCall("gdiplus\GdiplusShutdown","ptr",gr.tok)
        gr.tok := 0
    }
}

Pruebas sugeridas tras validar

Arrastra el slider rápido: debe oír un clic cada 3 unidades, sin retrasos ni ráfagas.
Pulsa Ctrl+Z en racha: el tono sube hasta el quinto nivel y baja suave al parar.
Pon [audio] enabled=0 en theme.ini: no debe sonar nada, ni Q ni W.
Pon el cursor en un panel y mantén el barrido más de 3 s: debe cortarse solo; Esc debe cancelar.
